CREATE TABLE Transaction (
Transaction_ID integer NOT NULL,
Transaction_Date date,
Transaction_Workstation varchar(20),
Transaction_Total_Amount numeric,
Transaction_Payment_Method varchar(20),
Employee_ID integer, 
Store_ID integer,  
Customer_ID integer, 
PRIMARY KEY(Transaction_ID),
FOREIGN KEY (Employee_ID) REFERENCES Employee(Employee_ID),
FOREIGN KEY (Store_ID) REFERENCES Store(Store_ID),
FOREIGN KEY (Customer_ID) REFERENCES Customer(Customer_ID));

CREATE TABLE Store(
Store_ID integer,
Store_Phone_Number varchar(20),
Store_Address varchar(255),
Store_Name varchar(20),
Store_Owner_Name varchar(50),
PRIMARY KEY (Store_ID));

CREATE TABLE Employee(
Employee_ID integer,
Employee_SSN varchar(20),
Employee_Address varchar(255),
Employee_Name varchar(20),
PRIMARY KEY (Employee_ID));

CREATE TABLE Product(
Product_ID integer,
Product_Type varchar(50) NOT NULL,
Product_Name varchar(40) NOT NULL,
Product_Price numeric,
Product_Unit_Of_Measure varchar(40),
PRIMARY KEY (Product_ID));

CREATE TABLE Includes(
Transaction_ID integer,
Product_ID integer,
Product_Quantity integer,
PRIMARY KEY (Transaction_ID, Product_ID),
FOREIGN KEY (Transaction_ID) REFERENCES Transaction (Transaction_ID),
FOREIGN KEY (Product_ID) REFERENCES Product (Product_ID));

CREATE TABLE Restock(
Restock_ID integer,
Store_ID integer,
Product_ID integer,
Restock_Quantity integer,
Restock_Date date,
PRIMARY KEY (Restock_ID),
FOREIGN KEY (Store_ID) REFERENCES Store (Store_ID),
FOREIGN KEY (Product_ID) REFERENCES Product (Product_ID));

CREATE TABLE Customer (
Customer_ID integer PRIMARY KEY,
Customer_Name VARCHAR(50) NOT NULL,
Customer_Membership VARCHAR(20) CHECK (Customer_Membership IN ('Gold', 'Silver', 'Bronze')),
Customer_Address varchar(100),
Customer_Age integer CHECK (Customer_Age > 0));
